// numbers -> array de inteiros, n -> tamanho do array
procedimento insertionSort(numbers, n) {
    para i = 2 até n
        valorAtual = numbers[i]
        j = i - 1
        
        enquanto j > 0 && numbers[j] > valorAtual {
            numbers[j + 1] = numbers[j]
            j = j - 1
        }   
            
        numbers[j + 1] = valorAtual
}
